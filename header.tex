% 산세리프도 한글 폰트로 (제목 등에 사용)

\titleformat{\subsubsection}{\normalsize\bfseries}{}{0em}{}

% 수식 주변 여백 최적화
\AtBeginDocument{%
  \abovedisplayskip=0.6em plus 2pt minus 2pt
  \belowdisplayskip=0.6em plus 2pt minus 2pt
  \abovedisplayshortskip=0.3em plus 2pt minus 2pt
